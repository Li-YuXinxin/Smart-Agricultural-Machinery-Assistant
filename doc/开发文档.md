## 一、项目概述

本项目是一个**智能植物养护助手**系统，包含 Python 后端服务和微信小程序前端两部分。核心功能包括：

| 功能模块 | 说明 |
|---------|------|
| 🌸 花卉识别 | 基于 ResNet-50 的花卉图像分类，支持 Top-5 预测及置信度展示 |
| 🧠 模型训练 | 用户上传花卉图片，在线微调 ResNet-50 模型，扩展可识别品种 |
| 📖 知识库管理 | 上传养护文档（PDF/DOCX/TXT），构建 RAG 向量知识库 |
| 💬 智能问答 | 基于 RAG + LLM（Qwen2.5）的植物养护 AI 顾问，流式输出 |

### 技术栈

**后端：**
- Python 3 + FastAPI（Web 框架）
- PyTorch + torchvision（深度学习推理与微调）
- LangChain + ChromaDB（RAG 向量检索）
- llama-cpp-python（本地 LLM 推理，Qwen2.5-1.5B GGUF）
- sentence-transformers（文本向量化，all-MiniLM-L6-v2）

**前端：**
- 微信小程序原生开发（Skyline 渲染引擎 + glass-easel 组件框架）
- WebSocket 实时训练状态推送
- SSE（Server-Sent Events）流式 AI 对话

```python
# WebSocket 连接管理器
class ConnectioConnnManager:
    def __init__(self):
        self.active_connections: list[WebSocket] = []

    async def connect(self, websocket: WebSocket):
        await websocket.accept()
        self.active_connections.append(websocket)

    def disconnect(self, websocket: WebSocket):
        if websocket in self.active_connections:
            self.active_connections.remove(websocket)

    async def broadcast(self, message: dict):
        """广播消息到所有连接的客户端"""
        tasks = [conn.send_json(message) for conn in self.active_connections]
        await asyncio.gather(*tasks, return_exceptions=True)

manager = ConnectioConnnManager()

def broadcast_status(status: dict):
    """从后台训练线程广播到主事件循环，再推送到前端"""
    loop = train_service._main_loop
    if loop and loop.is_running():
        asyncio.run_coroutine_threadsafe(manager.broadcast(status), loop)

train_service.set_broadcast_callback(broadcast_status)

# WebSocket 端点
@router.websocket("/ws")
async def websocket_endpoint(websocket: WebSocket):
    await manager.connect(websocket)
    await websocket.send_json(train_service.get_status())  # 发送当前状态
    try:
        while True:
            await websocket.receive_text()  # 保持连接
    except WebSocketDisconnect:
        manager.disconnect(websocket)

# 微调端点
@router.post("/finetune")
async def finetune_resnet50(request: Request):
    body = await request.json()
    label = body.get("label")
    clear_old = body.get("clearOld", False)

            shutil.rmtree(item) if item.is_dir() else item.unlink()

    class_dir.mkdir(parents=True, exist_ok=True)
    for idx, b64_data in enumerate(images_base64):
        b64_data = b64_data.split(",", 1)[-1]
        img_bytes = base64.b64decode(b64_data)
        img = Image.open(io.BytesIO(img_bytes))
        img.save(class_dir / f"{label}_{idx+1}.jpg")

    # 启动后台训练
```

### 3.8 图像分类服务 — `classify_service.py`

**功能：** 核心图像分类服务（单例模式）。包含 ResNet-50 模型加载、微软权重键名适配、余弦分类器、标签平滑损失、模型微调流程和图像预测。

#### 核心类：余弦分类器

用余弦相似度替代线性分类，提升小样本分类效果：

```python
class ConsineClassifier(nn.Module):
    def __init__(self, in_features, num_classes, scale=1.0):
        super().__init__()
        self.scale = nn.Parameter(torch.tensor(scale, dtype=torch.float))
        self.weight = nn.Parameter(torch.empty(in_features, num_classes))
        nn.init.kaiming_uniform_(self.weight, a=math.sqrt(5.0))

    def forward(self, x):
        weight_norm = F.normalize(self.weight, dim=1, p=2)
        x_norm = F.normalize(x, dim=1, p=2)
        return (x_norm @ weight_norm) * self.scale
```

#### 核心类：标签平滑交叉熵损失

防止模型过于自信，提升泛化能力：

```python
class LabelSmoothingCrossEntropy(nn.Module):
    def __init__(self, smoothing=0.1):
        super().__init__()
        self.smoothing = smoothing

    def forward(self, pred, target):
        n_classes = pred.size(1)
        one_hot = torch.zeros_like(pred).scatter(1, target.unsqueeze(1), 1)
        one_hot = one_hot * (1 - self.smoothing) + self.smoothing / n_classes
        log_prob = F.log_softmax(pred, dim=1)
        return -(one_hot * log_prob).sum(dim=1).mean()
```

#### 核心函数：微软权重键名适配

微软 ResNet-50 的 safetensors 权重键名与标准 torchvision 不同，需要正则映射：

```python
def adapt_timm_resnet_state_dict(state_dict):
    """微软键名 -> 标准 torchvision 键名"""
    new_state_dict = {}
    for k, v in state_dict.items():
        # embedder -> conv1 / bn1
        if k.startswith("resnet.embedder.embedder.convolution"):
            new_state_dict[f"conv1.{k.split('.')[-1]}"] = v
        # encoder.stages.X.layers.Y.layer.Z -> layer(X+1).Y.conv(Z+1)
        # 正则匹配主分支和捷径分支的卷积/归一化层
        m = pattern.match(k)
        if m:
            stage, block, idx = int(m.group(1)), int(m.group(2)), int(m.group(3))
            kind, suffix = m.group(4), m.group(5)
            layer = f"layer{stage+1}"
            prefix = "conv" if kind == "convolution" else "bn"
            new_state_dict[f"{layer}.{block}.{prefix}{idx+1}.{suffix}"] = v
    return new_state_dict
```

#### 核心方法：图像预测

```python
def predict(self, image_bytes, top_k=5):
    image = Image.open(io.BytesIO(image_bytes)).convert("RGB")
    input_tensor = self.transform(image).unsqueeze(0).to(self.device)
    with torch.no_grad():
        probs = torch.softmax(self.model(input_tensor), dim=1).cpu().numpy()[0]
    top_indices = np.argsort(probs)[-top_k:][::-1]
    top5 = [{"class": self.class_names[i], "confidence": float(probs[i])} for i in top_indices]
    # 置信度过滤: 低于阈值返回"未知类型"
    if top5[0]["confidence"] < settings.CONFIDENCE_THRESHOLD:
        return {"top1": "未知类型", "top1_confidence": 0.0, "top5": []}
    return {"top1": top5[0]["class"], "top1_confidence": top5[0]["confidence"], "top5": top5}
```

### 3.9 训练服务 — `train_service.py`

**功能：** 训练任务的生命周期管理服务（单例模式）。管理训练状态（idle/running/done/failed），在独立后台线程中执行微调，通过回调函数将进度推送到 WebSocket。

```python
class TrainService:
    def start_training(self, data_dir, epochs=10):
        """启动后台训练线程"""
        if self.status == "running": return False
        self.status = "running"
        self.stop_requested = False
        self.current_epoch = 0
        self.total_epochs = epochs
        self._broadcast()  # 广播"开始训练"状态
        # 启动后台线程
        self.current_task = threading.Thread(target=self._run, daemon=True)
        self.current_task.start()
        return True

    def _run(self):
        """后台线程: 调用 classify_service.finetune()"""
        def _on_epoch(current, total):
            self.current_epoch = current
            self.total_epochs = total
            self._broadcast()  # 每个 epoch 结束广播进度

        success = classify_service.finetune(
            data_dir=self._data_dir, epoch=self._epochs,
            stop_check=lambda e: self.stop_requested,  # 停止检查
            epoch_callback=_on_epoch)                   # 进度回调

        self.status = "done" if success else "failed"
        self._broadcast()  # 广播最终状态

    def get_status(self):
        return {"status": self.status, "result": self.result,
                "current_epoch": self.current_epoch, "total_epochs": self.total_epochs}

train_service = TrainService()
```

### 3.10 LLM 大语言模型服务 — `LLM_Service.py`

**功能：** 本地 LLM 推理服务（单例模式），使用 llama-cpp-python 加载 Qwen2.5-1.5B GGUF 量化模型。支持延迟加载和流式/非流式两种输出模式。

```python
class LLM_Service:
    def _load_model(self):
        """首次调用时延迟加载模型"""
        self.llm = Llama(
            model_path=self.model_path,
            n_ctx=settings.LLM_CONTEXT_SIZE,     # 上下文窗口 4096 tokens
            n_gpu_layers=settings.LLM_GPU_LAYERS, # GPU 层数（-1=全部）
            n_threads=settings.LLM_THREADS,        # 推理线程数
            verbose=False)
        self._loaded = True

    def chat(self, messages, temperature=0.7, max_tokens=512, stream=False):
        """对话接口，支持流式/非流式"""
        response = self.llm.create_chat_completion(
            messages=messages, temperature=temperature,
            max_tokens=max_tokens, stream=stream,
            stop=["<|im_end|>", "<|endoftext|>"])
        if stream:
            def stream_generator():
                for chunk in response:
                    content = chunk['choices'][0]['delta'].get('content', '')
                    if content: yield content
            return stream_generator()
        return response['choices'][0]['message']['content']

    def generate(self, user_prompt, system_prompt=None, temperature=0.7, max_tokens=512):
        """简化的 prompt 接口"""
        messages = []
        if system_prompt: messages.append({"role": "system", "content": system_prompt})
        messages.append({"role": "user", "content": user_prompt})
        return self.chat(messages, temperature, max_tokens)

llm_service = LLM_Service()
```

#### 核心方法：文档向量化

```python
def add_document(self, file_path):
    """加载文档 -> 文本分割 -> 向量化 -> 存入 ChromaDB"""
    # 根据后缀选择加载器: PyPDFLoader / Docx2txtLoader / TextLoader
    loader = PyPDFLoader(file_path) if suffix == ".pdf" else Docx2txtLoader(file_path)
    documents = loader.load()

    # 文本分割: 每段 500 字，重叠 50 字
    text_splitter = RecursiveCharacterTextSplitter(chunk_size=500, chunk_overlap=50)
    chunks = text_splitter.split_documents(documents)

    # 为每个段落添加来源元数据
    for chunk in chunks:
        chunk.metadata['source'] = file_path

    self.vectorstore.add_documents(chunks)
    self.retriever = self.vectorstore.as_retriever(search_kwargs={"k": 4})
    return {"success": True, "trunk_count": len(chunks)}
```

#### 核心方法：RAG 问答

```python
def query(self, question):
    """检索知识库 -> 拼接上下文 -> LLM 生成回答"""
    # 1. 向量检索最相关的 4 个段落
    docs = self.retriever.get_relevant_documents(question)

    # 2. 拼接上下文（上限 2000 字）
    context = "\n\n".join([doc.page_content for doc in docs[:settings.K]])
    if len(context) > 2000: context = context[:2000] + "..."

    # 3. 构建 prompt，调用 LLM 生成
    user_prompt = f"请根据以下知识库内容回答用户问题:\n知识库内容:\n{context}\n用户问题:\n{question}"
    answer = llm_service.generate(
        system_prompt=PromptTemplate.SAA_EXPERT,
        user_prompt=user_prompt, temperature=0.5, max_tokens=512)

    # 4. 提取来源文档（去重）
    source = list(dict.fromkeys([doc.metadata.get("source", "") for doc in docs]))
    return {"answer": answer, "source": source}

rag_service = RagService()
```

```javascript
  data: { result: null, isProcessing: false, history: [], showResult: false },

  // 拍照/选图
  takePhoto() {
    wx.chooseMedia({ count: 1, mediaType: ['image'], sourceType: ['album', 'camera'],
      success: (res) => {
        const tempFile = res.tempFiles[0]
        // 复制到 USER_DATA_PATH 持久化保存

        this.startRecognize()

    })
  },

  startRecognize() {
    // 1. 读取图片为 Base64
    fs.readFile({
      filePath: imagePath,
      success: (res) => {
        const base64 = wx.arrayBufferToBase64(res.data)
        const base64Data = `data:image/jpeg;base64,${base64}`

              ...item, confidenceText: (item.confidence * 100).toFixed(1) + '%'
            }))

            this.makeHistoryThumb(imagePath, (thumbPath) => {
              this.saveHistory({ name: res.data.top1, confidence: ..., thumbPath })
            })
          }
        })
      }
    })

    wx.setStorageSync('classifyHistory', history)
  }
```

### 4.4 训练新品种 — `pages/train/`

**功能：** 在线微调页面。选择图片 → 输入标签 → 提交训练 → WebSocket + HTTP 轮询实时展示进度。

```javascript
Page({
  data: { imageList: [], label: '', isTraining: false, status: { status: 'idle' } },

  onLoad() {
    this.initWebSocket()  // 建立 WebSocket 连接
    this.startPolling()   // HTTP 轮询兜底
  },

  // WebSocket 初始化（自动重连）
  initWebSocket() {
    const ws = wx.connectSocket({ url: `ws://${base}/api/train/ws` })
    ws.onMessage((res) => {
      const data = JSON.parse(res.data)
      this.setData({ status: data })
      if (data.status === 'done' || data.status === 'failed') {
        wx.vibrateShort({ type: 'heavy' })  // 震动提醒
      }
    })
    ws.onClose(() => { setTimeout(() => this.initWebSocket(), 3000) })  // 3s 自动重连
  },

  // HTTP 轮询兜底（每 2 秒）
  startPolling() {
    this._pollTimer = setInterval(() => {
      wx.request({
        url: `${app.globalData.apiBase}/api/train/status`,
        success: (res) => {
          this.setData({ status: res.data })
          if (res.data.status === 'done' || res.data.status === 'failed') {
            this.stopPolling()
          }
        }
      })
    }, 2000)
  },

  // 开始训练
  startFull() {
    // 读取所有图片为 Base64
    const tasks = this.data.imageList.map(item => new Promise((resolve) => {
      fs.readFile({ filePath: item.path,
        success: (res) => resolve(`data:image/jpeg;base64,${wx.arrayBufferToBase64(res.data)}`) })
    }))
    Promise.all(tasks).then((images) => {
      wx.request({
        url: `${app.globalData.apiBase}/api/train/finetune`,
        method: 'POST',
        data: { images, label: this.data.label.trim(), clearOld: this.data.clearOld },
        success: () => { this.startPolling() }  // 训练提交后启动轮询
      })
    })
  },

  // 停止训练
  stopTraining() {
    wx.request({ url: `${app.globalData.apiBase}/api/train/stop`, method: 'POST' })
  }
})
```

### 4.6 AI 养护顾问 — `pages/chat/`

**功能：** AI 智能问答页面，通过 SSE 流式接收后端回答，支持 Markdown 渲染、常见问题快捷入口、来源文档预览。

```javascript
Page({
  data: {
    messages: [], inputText: '', isStreaming: false,
    hotQuestions: ['绿萝叶子发黄怎么办？', '多肉多久浇一次水？', '室内养花适合什么光照？', '植物长虫了怎么处理？']
  },

  // 发送消息
  sendMessage() {
    const text = this.data.inputText.trim()
    if (!text || this.data.isStreaming) return

    // 添加用户消息 + AI 占位消息
    this.setData({
      messages: [...this.data.messages, { role: 'user', content: text, time: this.getTime() }],
      isStreaming: true
    })

    // SSE 流式请求
    const requestTask = wx.request({
      url: `${app.globalData.apiBase}/api/chat/stream`,
      method: 'POST', data: { message: text, temperature: 0.7, max_tokens: 512 },
      enableChunked: true,        // 启用分块传输
      responseType: 'arraybuffer' // 二进制缓冲区
    })

    // 接收数据块
    let buffer = ''
    requestTask.onChunkReceived((res) => {
      const str = this.uft8ArrayToString(new Uint8Array(res.data))
      buffer += str
      const lines = buffer.split('\n')
      buffer = lines.pop() || ''

      for (const line of lines) {
        if (line.startsWith('data:')) {
          const data = JSON.parse(line.substring(5).trim())
          if (data.done) {
            this.onDone()      // 传输完成
          } else if (data.chunk) {
            this.onChunk(data.chunk)  // 追加内容
          }
        }
      }
    })
  },

  // 接收数据块时实时更新
  onChunk(chunk) {
    const msgs = this.data.messages
    const last = msgs[msgs.length - 1]
    if (last.role === 'ai') {
      if (last.content === '思考中...') last.content = chunk
      else last.content += chunk
      last.htmlContent = this.mdToHtml(last.content)  // 实时 Markdown 渲染
      this.setData({ messages: msgs })
    }
  },

  // 轻量 Markdown -> HTML 转换
  mdToHtml(md) {
    let html = md.replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;')
    html = html.replace(/```(\w*)\n?([\s\S]*?)```/g, '<pre><code>$2</code></pre>')
    html = html.replace(/\*\*(.+?)\*\*/g, '<strong>$1</strong>')
    html = html.replace(/\*(.+?)\*/g, '<em>$1</em>')
    html = html.replace(/\n/g, '<br/>')
    return html
  },

  // UTF-8 ArrayBuffer -> 字符串（微信小程序不支持 TextDecoder）
  uft8ArrayToString(uint8Array) { /* ... 手动解码 UTF-8 字节 ... */ }
})
```
